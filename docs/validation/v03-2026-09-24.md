To reproduce on a running isolated backend, first use `docker port vv-v03-backend 8000/tcp` to find the published loopback port. For the recorded run it was `64176`:

```powershell
$wav = Join-Path $env:TEMP 'vigilvoice-v03-10s.wav'
ffmpeg -hide_banner -loglevel error -stream_loop 4 -i models/demo/genuine.wav -t 10 -ac 1 -ar 16000 -c:a pcm_s16le $wav
$times = 1..12 | ForEach-Object {
  $session = Invoke-RestMethod -Method Post -Uri 'http://127.0.0.1:64176/api/v1/sessions' -ContentType 'application/json' -Body '{}'
  $watch = [Diagnostics.Stopwatch]::StartNew()
  $result = Invoke-RestMethod -Method Post -Uri ("http://127.0.0.1:64176/api/v1/sessions/$($session.session_id)/audio") -Headers @{Authorization="Bearer $($session.session_token)"} -ContentType 'audio/wav' -InFile $wav
  $watch.Stop()
  if ($result.risk_state -ne 'ELEVATED') { throw "Unexpected risk $($result.risk_state)" }
  $watch.Elapsed.TotalMilliseconds
}
$sorted = @($times | Sort-Object)
"median=$((($sorted[5]+$sorted[6])/2)) ms; p95=$($sorted[[math]::Ceiling(.95*$sorted.Count)-1]) ms"
python scripts/test_client.py --server-uri ws://127.0.0.1:64176/api/v1/stream/ws --api-uri http://127.0.0.1:64176/api/v1 --duration-seconds 30
python scripts/test_client.py --server-uri ws://127.0.0.1:64176/api/v1/stream/ws --api-uri http://127.0.0.1:64176/api/v1 --duration-seconds 600
python scripts/test_client.py --server-uri ws://127.0.0.1:64176/api/v1/stream/ws --api-uri http://127.0.0.1:64176/api/v1 --audio-path models/demo/synthetic.wav
docker stats --no-stream --format '{{.Name}} {{.CPUPerc}} {{.MemUsage}}' vv-v03-backend vv-s02-db
```
